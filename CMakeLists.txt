cmake_minimum_required(VERSION 3.10)
project(local_monitor LANGUAGES CXX)
option(BUILD_GUI "Build Qt Widgets application" ON)
find_package(Threads REQUIRED)
add_library(monitor_core core/monitor.cpp core/devices.cpp core/linux_devices.cpp core/sampling.cpp core/actuators.cpp)
target_include_directories(monitor_core PUBLIC core)
set_target_properties(monitor_core PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_link_libraries(monitor_core PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(monitor_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(monitor_tests tests/monitor_tests.cpp)
set_target_properties(monitor_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_link_libraries(monitor_tests PRIVATE monitor_core)
add_test(NAME monitor_core COMMAND monitor_tests)
set_tests_properties(monitor_core PROPERTIES TIMEOUT 20)
add_executable(actuator_tests tests/actuator_tests.cpp)
set_target_properties(actuator_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_link_libraries(actuator_tests PRIVATE monitor_core)
add_test(NAME actuator_core COMMAND actuator_tests)
set_tests_properties(actuator_core PROPERTIES TIMEOUT 20)
if(BUILD_GUI)
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
    add_executable(smarthome app/main.cpp app/mainwindow.cpp app/mainwindow.h app/bridge.h)
    set_target_properties(smarthome PROPERTIES AUTOMOC ON CXX_STANDARD_REQUIRED YES)
    if(QT_VERSION_MAJOR GREATER 5)
        set_property(TARGET smarthome PROPERTY CXX_STANDARD 17)
    else()
        set_property(TARGET smarthome PROPERTY CXX_STANDARD 11)
    endif()
    target_link_libraries(smarthome PRIVATE monitor_core Qt${QT_VERSION_MAJOR}::Widgets)
    add_test(NAME qt_simulated_workflow COMMAND smarthome --simulate --smoke-test)
    set_tests_properties(qt_simulated_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
    add_test(NAME reject_implicit_simulation COMMAND smarthome --smoke-test)
    set_tests_properties(reject_implicit_simulation PROPERTIES WILL_FAIL TRUE TIMEOUT 5)
    add_test(NAME shutdown_output_failure COMMAND ${CMAKE_COMMAND}
        -DMONITOR_BINARY=$<TARGET_FILE:smarthome>
        -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/expect_shutdown_failure.cmake)
    set_tests_properties(shutdown_output_failure PROPERTIES TIMEOUT 15)
    add_test(NAME shutdown_servo_failure COMMAND ${CMAKE_COMMAND}
        -DMONITOR_BINARY=$<TARGET_FILE:smarthome>
        -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/expect_servo_shutdown_failure.cmake)
    set_tests_properties(shutdown_servo_failure PROPERTIES TIMEOUT 15)
endif()
